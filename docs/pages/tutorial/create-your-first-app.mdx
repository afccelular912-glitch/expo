// App.js — HUBA Starter App
import React from "react";
import { NavigationContainer } from "@react-navigation/native";
import { createBottomTabNavigator } from "@react-navigation/bottom-tabs";
import { View, Text, TouchableOpacity, StyleSheet, StatusBar, ImageBackground } from "react-native";
import { LinearGradient } from "expo-linear-gradient";
import { Ionicons } from "@expo/vector-icons";

const Tab = createBottomTabNavigator();

// --- Screens ---
function SplashScreen({ navigation }) {
  return (
    <LinearGradient colors={["#1A4BFF", "#6B2BFF"]} style={styles.container}>
      <StatusBar barStyle="light-content" />
      <Text style={styles.logo}>HUBA</Text>
      <Text style={styles.subtitle}>Converse. Crie. Conecte-se.</Text>
      <TouchableOpacity style={styles.button} onPress={() => navigation.replace("Main")}>
        <Text style={styles.buttonText}>Entrar</Text>
      </TouchableOpacity>
    </LinearGradient>
  );
}

function HomeScreen() {
  return (
    <View style={styles.screen}>
      <Text style={styles.title}>🎧 Salas de Voz Ativas</Text>
      <Text style={styles.card}>#Amigos Jogando</Text>
      <Text style={styles.card}>#Bate-Papo Geral</Text>
      <Text style={styles.card}>#Música e Relax</Text>
    </View>
  );
}

function ChatScreen() {
  return (
    <View style={styles.screen}>
      <Text style={styles.title}>💬 Chat HUBA</Text>
      <Text style={styles.text}>Em breve: mensagens e emojis personalizados.</Text>
    </View>
  );
}

function VoiceScreen() {
  return (
    <View style={styles.screen}>
      <Text style={styles.title}>🔊 Sala de Voz</Text>
      <Text style={styles.text}>Em breve: chamadas em tempo real com Agora SDK.</Text>
    </View>
  );
}

function ProfileScreen() {
  return (
    <View style={styles.screen}>
      <Text style={styles.title}>👤 Meu Perfil</Text>
      <Text style={styles.text}>Nome de usuário, status e configurações.</Text>
    </View>
  );
}

// --- Tab Navigation ---
function MainTabs() {
  return (
    <Tab.Navigator
      screenOptions={{
        headerShown: false,
        tabBarStyle: { backgroundColor: "#0E0E12", borderTopColor: "#222" },
        tabBarActiveTintColor: "#6B2BFF",
      }}
    >
      <Tab.Screen
        name="Home"
        component={HomeScreen}
        options={{ tabBarIcon: ({ color, size }) => <Ionicons name="home" color={color} size={size} /> }}
      />
      <Tab.Screen
        name="Chat"
        component={ChatScreen}
        options={{ tabBarIcon: ({ color, size }) => <Ionicons name="chatbubbles" color={color} size={size} /> }}
      />
      <Tab.Screen
        name="Voz"
        component={VoiceScreen}
        options={{ tabBarIcon: ({ color, size }) => <Ionicons name="mic" color={color} size={size} /> }}
      />
      <Tab.Screen
        name="Perfil"
        component={ProfileScreen}
        options={{ tabBarIcon: ({ color, size }) => <Ionicons name="person" color={color} size={size} /> }}
      />
    </Tab.Navigator>
  );
}

// --- Root Navigation ---
export default function App() {
  const [loaded, setLoaded] = React.useState(true);

  return (
    <NavigationContainer>
      {loaded ? <SplashScreen navigation={{ replace: () => setLoaded(false) }} /> : <MainTabs />}
    </NavigationContainer>
  );
}

    textAlign: "center",

  },
});
